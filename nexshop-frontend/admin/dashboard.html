                Cara toko lain memverifikasi payload dari NexShop
                
// Node.js -- di sisi penerima
const crypto = require("crypto");

app.post("/api/callback/tokovoucher", express.raw({ type: "*/*" }), (req, res) => {
  const raw = req.body.toString("utf8");
  const ts  = req.headers["x-nexshop-timestamp"];
  const sig = req.headers["x-nexshop-signature"]; // "sha256=..."

  const expected = "sha256=" + crypto
    .createHmac("sha256", process.env.NEXSHOP_WEBHOOK_SECRET)
    .update(ts + "." + raw)
    .digest("hex");

  if (sig !== expected) return res.status(401).send("bad signature");
  // Tolak payload yang lebih tua dari 5 menit (anti replay)
  if (Math.abs(Date.now() / 1000 - Number(ts)) > 300) return res.status(408).send("stale");

  const body = JSON.parse(raw);
  const trx  = body.data; // payload asli TokoVoucher: ref_id, status, sn, ...
  // ... update status transaksi di sistem kamu ...

  res.status(200).json({ ok: true }); // WAJIB 2xx, kalau enggak NexShop retry
});
